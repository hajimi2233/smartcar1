cmake_minimum_required(VERSION 2.8.3)
project(smartcar_navigation)
find_package(catkin REQUIRED)
catkin_package()
catkin_install_python(PROGRAMS
  scripts/region_editor.py
  scripts/single_goal_nav.py
  scripts/multi_goal_nav.py
  scripts/send_single_goal.py
  scripts/terminal_teleop.py
  scripts/waypoint_nav.py
  scripts/laser_noise.py
  scripts/imu_bridge.py
  scripts/scan_in_map.py
  scripts/scan_match_localizer.py
  scripts/reset_to_start.py
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
# Helper module must be importable, not a catkin executable relay wrapper.
file(MAKE_DIRECTORY ${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION})
configure_file(scripts/ackermann_core.py
  ${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION}/ackermann_core.py COPYONLY)
install(FILES scripts/ackermann_core.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})

configure_file(scripts/region_geometry.py ${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION}/region_geometry.py COPYONLY)
install(FILES scripts/region_geometry.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})

configure_file(scripts/nav_obstacles.py ${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION}/nav_obstacles.py COPYONLY)
install(FILES scripts/nav_obstacles.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})

configure_file(scripts/local_planner.py ${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION}/local_planner.py COPYONLY)
install(FILES scripts/local_planner.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})

configure_file(scripts/actuator_model.py ${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION}/actuator_model.py COPYONLY)
install(FILES scripts/actuator_model.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})

configure_file(scripts/path_tracking.py ${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION}/path_tracking.py COPYONLY)
install(FILES scripts/path_tracking.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
